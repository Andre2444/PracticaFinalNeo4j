Cargar nodos desde CSV

// Usuarios
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Andre2444/PracticaFinalNeo4j/main/usuarios.csv' AS row
MERGE (u:Usuario {usuarioId: toInteger(row.usuarioId)})
SET u.nombre = row.nombre,
    u.edad = toInteger(row.edad),
    u.sexo = row.sexo,
    u.email = row.email,
    u.fechaNacimiento = row.fechaNacimiento;

// Cursos
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Andre2444/PracticaFinalNeo4j/main/cursos.csv' AS row

SET c.titulo = row.titulo,
    c.descripcion = row.descripcion,
    c.nivel = row.nivel,
    c.anioPublicacion = toInteger(row.anioPublicacion),
    c.calificacionPromedio = toFloat(row.calificacionPromedio);

Cargar relaciones desde CSV

// (:Usuario)-[:VIO]->(:Leccion)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Andre2444/PracticaFinalNeo4j/main/rel_vio.csv' AS row
MATCH (origen:Usuario {usuarioId: toInteger(row.usuarioId)})
MATCH (destino:Leccion {leccionId: toInteger(row.leccionId)})
MERGE (origen)-[r:VIO]->(destino)
SET r.fecha = row.fecha,
    r.minutosReproducidos = toInteger(row.minutosReproducidos),
    r.finalizado = toBoolean(row.finalizado);

// (:Usuario)-[:LE_GUSTA]->(:Curso)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/Andre2444/PracticaFinalNeo4j/main/rel_le_gusta.csv' AS row
MATCH (origen:Usuario {usuarioId: toInteger(row.usuarioId)})
MATCH (destino:Curso {cursoId: toInteger(row.cursoId)})
MERGE (origen)-[r:LE_GUSTA]->(destino)
SET r.valoracion = toInteger(row.valoracion),
    r.fecha = row.fecha;

// Calcular calificacionPromedio de Curso derivada de las valoraciones de los usuarios
MATCH (:Usuario)-[r:LE_GUSTA]->(c:Curso)
WITH c, round(avg(r.valoracion) * 10.0) / 10.0 AS promedio
SET c.calificacionPromedio = promedio;

Consultas

// 1. ¿Cuál es la categoría que más ha cursado un usuario?
MATCH (u:Usuario {usuarioId: 1})-[:INSCRITO_EN]->(c:Curso)-[:PERTENECE_A]->(cat:Categoria)
RETURN u.nombre AS usuario, cat.nombre AS categoria, COUNT(c) AS totalCursos
ORDER BY totalCursos DESC
LIMIT 1;

// 2. ¿Qué lecciones de un curso específico ha visto un usuario inscrito?
MATCH (u:Usuario {usuarioId: 1})-[:INSCRITO_EN]->(c:Curso {cursoId: 1})-[:ESTA_COMPUESTO]->(m:Modulo)-[:CONTIENE]->(l:Leccion)
MATCH (u)-[v:VIO]->(l)
RETURN u.nombre AS usuario, c.titulo AS curso, l.titulo AS leccion, v.fecha AS fecha, v.minutosReproducidos AS minutosReproducidos, v.finalizado AS finalizado
ORDER BY m.numero, l.numero;

// 3. ¿Cuál es el curso más popular en un país determinado?
MATCH (p:Pais {nombre: 'Colombia'})<-[:RESIDE_EN]-(u:Usuario)-[:INSCRITO_EN]->(c:Curso)
RETURN p.nombre AS pais, c.titulo AS curso, COUNT(u) AS totalInscripciones
ORDER BY totalInscripciones DESC
LIMIT 1;

// 4. ¿Qué cursos están disponibles en más de tres países?
MATCH (c:Curso)-[:DISPONIBLE_EN]->(p:Pais)
WITH c, COUNT(DISTINCT p) AS cantidadPaises
WHERE cantidadPaises > 3
RETURN c.titulo AS curso, cantidadPaises
ORDER BY cantidadPaises DESC;

// 5. ¿Qué curso popular en el país de residencia de un usuario puede recomendársele, si aún no está inscrito?
MATCH (u:Usuario {usuarioId: 1})-[:RESIDE_EN]->(p:Pais)<-[:RESIDE_EN]-(otro:Usuario)-[:INSCRITO_EN]->(c:Curso)
WHERE otro <> u AND NOT (u)-[:INSCRITO_EN]->(c)
RETURN c.titulo AS cursoRecomendado, p.nombre AS pais, COUNT(DISTINCT otro) AS inscripcionesEnPais
ORDER BY inscripcionesEnPais DESC
LIMIT 1;

// 6. ¿Qué cursos se pueden recomendar a un usuario según las categorías que ha cursado previamente?
MATCH (u:Usuario {usuarioId: 1})-[:INSCRITO_EN]->(:Curso)-[:PERTENECE_A]->(cat:Categoria)<-[:PERTENECE_A]-(recomendado:Curso)
WHERE NOT (u)-[:INSCRITO_EN]->(recomendado)
RETURN DISTINCT recomendado.titulo AS cursoRecomendado, cat.nombre AS categoria;

// 7. ¿Qué cursos han cursado otros usuarios con historial de inscripción similar, y que el usuario aún no ha cursado?
MATCH (u:Usuario {usuarioId: 1})-[:INSCRITO_EN]->(comun:Curso)<-[:INSCRITO_EN]-(similar:Usuario)-[:INSCRITO_EN]->(recomendado:Curso)
WHERE u <> similar AND NOT (u)-[:INSCRITO_EN]->(recomendado)
RETURN recomendado.titulo AS cursoRecomendado, COUNT(DISTINCT comun) AS cursosEnComun, COUNT(DISTINCT similar) AS usuariosSimilares
ORDER BY cursosEnComun DESC, usuariosSimilares DESC;
